USE ql_thongtinchibo;

CREATE TABLE chibo (
    stt INT AUTO_INCREMENT,                     -- Số thứ tự (Tự động tăng)
    ten_chibo VARCHAR(100) NOT NULL,            -- Tên Chi Bộ (Dùng làm Khóa chính để liên kết)
    tong_so_dang_vien INT DEFAULT 0,           -- Tổng số Đảng Viên
    chinh_thuc INT DEFAULT 0,                  -- Số đảng viên Chính thức
    du_bi INT DEFAULT 0,                       -- Số đảng viên Dự Bị
    PRIMARY KEY (ten_chibo),                   -- Chọn tên chi bộ làm khóa chính
    UNIQUE (stt)                               -- Đảm bảo STT không bị trùng lặp
);

ALTER TABLE chibo
ADD COLUMN cap_to_chuc VARCHAR(100) 
    CHARACTER SET utf8mb4 
    COLLATE utf8mb4_unicode_ci 
    DEFAULT NULL;



CREATE TABLE tai_khoan (
    user VARCHAR(50) NOT NULL PRIMARY KEY,
    pass VARCHAR(255) NOT NULL,
    hoten VARCHAR(100),
    chibo VARCHAR(100),
    chucvu VARCHAR(100),
    role VARCHAR(50),
    -- Tạo liên kết: cột chibo của bảng này phải tồn tại trong cột ten_chibo của bảng chibo
    FOREIGN KEY (chibo) REFERENCES chibo(ten_chibo) ON UPDATE CASCADE ON DELETE SET NULL
);
